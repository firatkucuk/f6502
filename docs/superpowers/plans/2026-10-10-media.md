# Media (sub-project 6) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Load programs from D64/T64/P00/PRG disk images through a KERNAL-level trap (with D64 write support and export) and from TAP files through a pulse-level datasette, in Node and in the browser.

**Architecture:** `src/machine/d64.ts` is the image model; `container.ts` holds the read-only T64/P00/PRG sources and the BASIC directory listing; `drive.ts` implements channels, file commands and the DOS command channel on top of an image; `kernal-traps.ts` intercepts the documented KERNAL indirect vectors at instruction boundaries and serves device 8-11 itself; `datasette.ts` plays TAP pulses into CIA1's FLAG pin gated by the 6510 port motor/sense bits. `browser/` adds attach/export/autostart and warp pacing. `core/` is untouched.

**Tech Stack:** TypeScript 6, Vitest 5, tsup, ESLint boundary rules, Prettier.

**Spec:** `docs/superpowers/specs/2026-10-10-media-design.md` (binding).

## Global Constraints

- No ROM, disk image or GPL file is vendored; reference programs and images are fetched, pinned, and git-ignored (`.testdata/`).
- `src/core` and `src/machine` must not reference DOM or Web APIs. `core/` is not modified.
- Public API additions only: `C64.drive(n)`, `C64.tape`, `C64.warp`, `Player.mountDisk/insertTape/exportDisk`, `StartOptions.disk/tape/autostart`.
- A trap costs one array lookup per instruction (`trapMap[pc]`); the loop must stay allocation-free.

## Review Focus

- A BASIC `LOAD`/`SAVE`/`OPEN` for a device we do not serve (printer 4, tape 1, unmounted drive 8) must behave exactly as before (device-not-present error from the ROM).
- `LOAD"*",8` on an empty image, a name longer than 16 characters, a zero-length file, and a file whose chain loops must not hang the machine or throw out of `runFrame`.
- Writing to a full disk, to a read-only source (T64/PRG), or with a duplicate name must report the DOS error and leave the image bytes unchanged.
- A malformed D64/T64/TAP (wrong size, truncated, junk) must be rejected with a clear error at mount/insert time, never at LOAD time.
- A tape that ends, is stopped or has the motor switched off mid-load must not leave the machine in warp or the SID muted.
- Exporting a mounted D64 returns a copy: later writes must not mutate a previously exported array.

```ts
export const BLOCK = 256;
const DIR_TRACK = 18;
const FILE_INTERLEAVE = 10;
const DIR_INTERLEAVE = 3;

export class DiskError extends Error {
  constructor(
    readonly code: number,
    message: string,
  ) {
    super(message);
    this.name = 'DiskError';
  }
}

export function sectorsInTrack(track: number): number {
  return track <= 17 ? 21 : track <= 24 ? 19 : track <= 30 ? 18 : 17;
}

export function petscii(s: string): Uint8Array {
  return Uint8Array.from(s.toUpperCase(), (c) => c.charCodeAt(0) & 0xff);
}

/** Trailing-$A0-trimmed bytes to text (shifted letters $C1-$DA map to A-Z). */
export function toText(name: Uint8Array): string {
  let s = '';
  for (const b of name) {
    if (b === 0xa0) break;
    s += String.fromCharCode(b >= 0xc1 && b <= 0xda ? b - 0x80 : b);
  }
  return s;
}

export interface DirEntry {
  slot: number;
  name: Uint8Array;
  /** 0 DEL, 1 SEQ, 2 PRG, 3 USR, 4 REL */
  type: number;
  closed: boolean;
  locked: boolean;
  track: number;
  sector: number;
  blocks: number;
}

const SIZES: Record<number, { tracks: 35 | 40; errors: boolean }> = {
  174848: { tracks: 35, errors: false },
  175531: { tracks: 35, errors: true },
  196608: { tracks: 40, errors: false },
  197376: { tracks: 40, errors: true },
};

export class D64 {
  readonly tracks: 35 | 40;
  private readonly data: Uint8Array;
  private readonly errors: Uint8Array | null;
  private readonly trackStart: number[] = [0];

  constructor(bytes: Uint8Array) {
    const info = SIZES[bytes.length];
    if (!info) throw new DiskError(74, `not a D64 image (size ${bytes.length})`);
    this.tracks = info.tracks;
    let total = 0;
    for (let t = 1; t <= this.tracks; t++) {
      this.trackStart[t] = total;
      total += sectorsInTrack(t);
    }
    this.data = bytes.slice(0, total * BLOCK);
    this.errors = info.errors ? bytes.slice(total * BLOCK) : null;
  }

  static format(name: string, id: string, tracks: 35 | 40 = 35): D64 {
    let total = 0;
    for (let t = 1; t <= tracks; t++) total += sectorsInTrack(t);
    const d = new D64(new Uint8Array(total * BLOCK));
    d.format(name, id);
    return d;
  }

  /** Reinitialise the BAM and the directory; keeps the old id when `id` is omitted. */
  format(name: string, id?: string): void {
    const keepId = id === undefined ? this.diskId().slice() : petscii(id);
    this.data.fill(0);
    const bam = this.sector(DIR_TRACK, 0);
    bam[0] = DIR_TRACK;
    bam[1] = 1;
    bam[2] = 0x41;
    bam.fill(0xa0, 0x90, 0xab);
    this.setHeader(name, keepId);
    for (let t = 1; t <= this.tracks; t++) {
      for (let s = 0; s < sectorsInTrack(t); s++) this.setFree(t, s, true);
    }
    this.setFree(DIR_TRACK, 0, false);
    this.setFree(DIR_TRACK, 1, false);
    const dir = this.sector(DIR_TRACK, 1);
    dir[0] = 0;
    dir[1] = 0xff;
  }

  private setHeader(name: string, id: Uint8Array): void {
    const bam = this.sector(DIR_TRACK, 0);
    bam.fill(0xa0, 0x90, 0xab);
    const n = petscii(name).slice(0, 16);
    bam.set(n, 0x90);
    bam[0xa2] = id[0] ?? 0x30;
    bam[0xa3] = id[1] ?? 0x30;
    bam[0xa5] = 0x32;
    bam[0xa6] = 0x41;
  }

  diskName(): Uint8Array {
    return this.sector(DIR_TRACK, 0).slice(0x90, 0xa0);
  }

  diskId(): Uint8Array {
    return this.sector(DIR_TRACK, 0).slice(0xa2, 0xa4);
  }

  private sector(track: number, sector: number): Uint8Array {
    if (track < 1 || track > this.tracks || sector < 0 || sector >= sectorsInTrack(track)) {
      throw new DiskError(66, `illegal track and sector ${track}/${sector}`);
    }
    const o = (this.trackStart[track]! + sector) * BLOCK;
    return this.data.subarray(o, o + BLOCK);
  }

  // ---- BAM ---------------------------------------------------------------

  private bamEntry(track: number): number {
    const base = this.trackStart[DIR_TRACK]! * BLOCK;
    return base + (track <= 35 ? 4 + 4 * (track - 1) : 0xc0 + 4 * (track - 36));
  }

  isFree(track: number, sector: number): boolean {
    const o = this.bamEntry(track);
    return ((this.data[o + 1 + (sector >> 3)]! >> (sector & 7)) & 1) === 1;
  }

  private setFree(track: number, sector: number, free: boolean): void {
    const o = this.bamEntry(track);
    const idx = o + 1 + (sector >> 3);
    const bit = 1 << (sector & 7);
    const was = (this.data[idx]! & bit) !== 0;
    if (free === was) return;
    if (free) {
      this.data[idx]! |= bit;
      this.data[o]!++;
    } else {
      this.data[idx]! &= ~bit;
      this.data[o]!--;
    }
  }

  freeBlocks(): number {
    let n = 0;
    for (let t = 1; t <= this.tracks; t++) if (t !== DIR_TRACK) n += this.data[this.bamEntry(t)]!;
    return n;
  }

  /** Tracks ordered by distance from the directory track: 17, 19, 16, 20, ... */
  private trackOrder(): number[] {
    const order: number[] = [];
    for (let d = 1; d < this.tracks; d++) {
      if (DIR_TRACK - d >= 1) order.push(DIR_TRACK - d);
      if (DIR_TRACK + d <= this.tracks) order.push(DIR_TRACK + d);
    }
    return order;
  }

  private findOnTrack(track: number, from: number): number {
    const n = sectorsInTrack(track);
    for (let i = 0; i < n; i++) {
      const s = (from + i) % n;
      if (this.isFree(track, s)) return s;
    }
    return -1;
  }

  /** Next data sector: stays on the current track while it has room, else the nearest track. */
  private allocData(prev: { t: number; s: number } | null): { t: number; s: number } | null {
    if (prev) {
      const s = this.findOnTrack(prev.t, (prev.s + FILE_INTERLEAVE) % sectorsInTrack(prev.t));
      if (s >= 0) return { t: prev.t, s };
    }
    for (const t of this.trackOrder()) {
      const s = this.findOnTrack(t, 0);
      if (s >= 0) return { t, s };
    }
    return null;
  }

  // ---- directory -----------------------------------------------------------

  /** Directory sectors in chain order (guarded against loops). */
  private dirSectors(): Array<{ t: number; s: number }> {
    const out: Array<{ t: number; s: number }> = [];
    const seen = new Set<number>();
    let t = DIR_TRACK;
    let s = 1;
    while (t !== 0 && out.length < 64) {
      const key = t * 64 + s;
      if (seen.has(key)) break;
      seen.add(key);
      let sec: Uint8Array;
      try {
        sec = this.sector(t, s);
      } catch {
        break;
      }
      out.push({ t, s });
      t = sec[0]!;
      s = sec[1]!;
    }
    return out;
  }

  entries(): DirEntry[] {
    const out: DirEntry[] = [];
    let slot = 0;
    for (const { t, s } of this.dirSectors()) {
      const sec = this.sector(t, s);
      for (let i = 0; i < 8; i++, slot++) {
        const o = i * 32;
        const typeByte = sec[o + 2]!;
        if (typeByte === 0) continue;
        out.push({
          slot,
          name: this.trimName(sec.subarray(o + 5, o + 21)),
          type: typeByte & 7,
          closed: (typeByte & 0x80) !== 0,
          locked: (typeByte & 0x40) !== 0,
          track: sec[o + 3]!,
          sector: sec[o + 4]!,
          blocks: sec[o + 28]! | (sec[o + 29]! << 8),
        });
      }
    }
    return out;
  }

  private trimName(raw: Uint8Array): Uint8Array {
    let n = raw.length;
    while (n > 0 && raw[n - 1] === 0xa0) n--;
    return raw.slice(0, n);
  }

  private slotRef(slot: number): { sec: Uint8Array; o: number } | null {
    const sectors = this.dirSectors();
    const s = sectors[slot >> 3];
    if (!s) return null;
    return { sec: this.sector(s.t, s.s), o: (slot & 7) * 32 };
  }

  private freeSlot(): { sec: Uint8Array; o: number; slot: number } {
    const sectors = this.dirSectors();
    for (let si = 0; si < sectors.length; si++) {
      const sec = this.sector(sectors[si]!.t, sectors[si]!.s);
      for (let i = 0; i < 8; i++) if (sec[i * 32 + 2] === 0) return { sec, o: i * 32, slot: si * 8 + i };
    }
    // grow the directory: new sector on the directory track, interleave 3
    const last = sectors[sectors.length - 1]!;
    const s = this.findOnTrack(DIR_TRACK, (last.s + DIR_INTERLEAVE) % sectorsInTrack(DIR_TRACK));
    if (s < 0) throw new DiskError(72, 'disk full (directory)');
    this.setFree(DIR_TRACK, s, false);
    this.sector(last.t, last.s)[0] = DIR_TRACK;
    this.sector(last.t, last.s)[1] = s;
    const sec = this.sector(DIR_TRACK, s);
    sec.fill(0);
    sec[1] = 0xff;
    return { sec, o: 0, slot: sectors.length * 8 };
  }

  // ---- files ---------------------------------------------------------------

  readFile(e: DirEntry): Uint8Array {
    const chunks: Uint8Array[] = [];
    const seen = new Set<number>();
    let t = e.track;
    let s = e.sector;
    let total = 0;
    while (t !== 0 && total < BLOCK * 800) {
      const key = t * 64 + s;
      if (seen.has(key)) break;
      seen.add(key);
      let sec: Uint8Array;
      try {
        sec = this.sector(t, s);
      } catch {
        break;
      }
      const nt = sec[0]!;
      const ns = sec[1]!;
      const chunk = nt !== 0 ? sec.subarray(2, 256) : sec.subarray(2, Math.max(2, ns) + 1);
      chunks.push(chunk);
      total += chunk.length;
      t = nt;
      s = ns;
    }
    const out = new Uint8Array(total);
    let o = 0;
    for (const c of chunks) {
      out.set(c, o);
      o += c.length;
    }
    return out;
  }

  private find(name: Uint8Array): DirEntry | undefined {
    return this.entries().find(
      (e) => e.name.length === name.length && e.name.every((b, i) => b === name[i]),
    );
  }

  writeFile(name: Uint8Array, type: number, data: Uint8Array): DirEntry {
    const trimmed = this.trimName(name.slice(0, 16));
    if (trimmed.length === 0) throw new DiskError(34, 'syntax error: empty name');
    if (this.find(trimmed)) throw new DiskError(63, 'file exists');
    const need = Math.max(1, Math.ceil(data.length / 254));
    if (need > this.freeBlocks()) throw new DiskError(72, 'disk full');
    // dry run of the directory slot so a failure cannot leave blocks allocated
    const slotInfo = this.freeSlot();
    const chain: Array<{ t: number; s: number }> = [];
    let prev: { t: number; s: number } | null = null;
    for (let i = 0; i < need; i++) {
      const next = this.allocData(prev);
      if (!next) {
        for (const c of chain) this.setFree(c.t, c.s, true);
        throw new DiskError(72, 'disk full');
      }
      this.setFree(next.t, next.s, false);
      chain.push(next);
      prev = next;
    }
    for (let i = 0; i < chain.length; i++) {
      const sec = this.sector(chain[i]!.t, chain[i]!.s);
      sec.fill(0);
      const chunk = data.subarray(i * 254, (i + 1) * 254);
      sec.set(chunk, 2);
      if (i < chain.length - 1) {
        sec[0] = chain[i + 1]!.t;
        sec[1] = chain[i + 1]!.s;
      } else {
        sec[0] = 0;
        sec[1] = chunk.length + 1;
      }
    }
    const { sec, o } = slotInfo;
    sec.fill(0, o + 3, o + 32);
    sec[o + 2] = 0x80 | (type & 7);
    sec[o + 3] = chain[0]!.t;
    sec[o + 4] = chain[0]!.s;
    sec.fill(0xa0, o + 5, o + 21);
    sec.set(trimmed, o + 5);
    sec[o + 28] = chain.length & 0xff;
    sec[o + 29] = chain.length >> 8;
    return {
      slot: slotInfo.slot,
      name: trimmed,
      type: type & 7,
      closed: true,
      locked: false,
      track: chain[0]!.t,
      sector: chain[0]!.s,
      blocks: chain.length,
    };
  }

  scratch(e: DirEntry): void {
    const seen = new Set<number>();
    let t = e.track;
    let s = e.sector;
    while (t !== 0 && !seen.has(t * 64 + s)) {
      seen.add(t * 64 + s);
      let sec: Uint8Array;
      try {
        sec = this.sector(t, s);
      } catch {
        break;
      }
      this.setFree(t, s, true);
      t = sec[0]!;
      s = sec[1]!;
    }
    const ref = this.slotRef(e.slot);
    if (ref) ref.sec[ref.o + 2] = 0;
  }

  rename(e: DirEntry, name: Uint8Array): void {
    const ref = this.slotRef(e.slot);
    if (!ref) return;
    ref.sec.fill(0xa0, ref.o + 5, ref.o + 21);
    ref.sec.set(this.trimName(name.slice(0, 16)), ref.o + 5);
  }

  /** Rebuild the BAM from the directory and the file chains. */
  validate(): void {
    for (let t = 1; t <= this.tracks; t++) {
      for (let s = 0; s < sectorsInTrack(t); s++) this.setFree(t, s, true);
    }
    this.setFree(DIR_TRACK, 0, false);
    for (const d of this.dirSectors()) this.setFree(d.t, d.s, false);
    for (const e of this.entries()) {
      const seen = new Set<number>();
      let t = e.track;
      let s = e.sector;
      while (t !== 0 && !seen.has(t * 64 + s)) {
        seen.add(t * 64 + s);
        let sec: Uint8Array;
        try {
          sec = this.sector(t, s);
        } catch {
          break;
        }
        this.setFree(t, s, false);
        t = sec[0]!;
        s = sec[1]!;
      }
    }
  }

  export(): Uint8Array {
    if (!this.errors) return this.data.slice();
    const out = new Uint8Array(this.data.length + this.errors.length);
    out.set(this.data, 0);
    out.set(this.errors, this.data.length);
    return out;
  }
}
```

import { D64, DiskError, petscii, type DirEntry } from './d64';

export interface SourceFile {
  name: Uint8Array;
  type: number;
  blocks: number;
  closed: boolean;
  locked: boolean;
}

export interface FileSource {
  readonly writable: boolean;
  label(): { name: Uint8Array; id: Uint8Array };
  list(): SourceFile[];
  /** First file whose name matches the pattern (`*`, `?`); the data is a PRG (address first). */
  read(name: Uint8Array): { file: SourceFile; data: Uint8Array } | null;
  blocksFree(): number;
}

/** CBM DOS matching: `?` matches one character, `*` matches the rest. */
export function matchName(pattern: Uint8Array, name: Uint8Array): boolean {
  for (let i = 0; i < pattern.length; i++) {
    const p = pattern[i]!;
    if (p === 0x2a) return true;
    if (i >= name.length) return false;
    if (p !== 0x3f && p !== name[i]) return false;
  }
  return pattern.length === name.length;
}

export class D64Source implements FileSource {
  readonly writable = true;
  constructor(readonly image: D64) {}

  label() {
    return { name: this.image.diskName(), id: this.image.diskId() };
  }

  private mapped(): Array<{ file: SourceFile; entry: DirEntry }> {
    return this.image.entries().map((entry) => ({
      entry,
      file: {
        name: entry.name,
        type: entry.type,
        blocks: entry.blocks,
        closed: entry.closed,
        locked: entry.locked,
      },
    }));
  }

  list(): SourceFile[] {
    return this.mapped().map((m) => m.file);
  }

  read(name: Uint8Array) {
    const hit = this.mapped().find((m) => matchName(name, m.file.name));
    return hit ? { file: hit.file, data: this.image.readFile(hit.entry) } : null;
  }

  blocksFree(): number {
    return this.image.freeBlocks();
  }
}

const NAME_PAD = 0x20;

function trimSpaces(raw: Uint8Array): Uint8Array {
  let n = raw.length;
  while (n > 0 && (raw[n - 1] === NAME_PAD || raw[n - 1] === 0x00)) n--;
  return raw.slice(0, n);
}

class ListSource implements FileSource {
  readonly writable = false;
  constructor(
    private readonly files: Array<{ file: SourceFile; data: Uint8Array }>,
    private readonly title: string,
  ) {}

  label() {
    return { name: petscii(this.title).slice(0, 16), id: petscii('00') };
  }

  list(): SourceFile[] {
    return this.files.map((f) => f.file);
  }

  read(name: Uint8Array) {
    return this.files.find((f) => matchName(name, f.file.name)) ?? null;
  }

  blocksFree(): number {
    return 0;
  }
}

function blocksOf(len: number): number {
  return Math.max(1, Math.ceil(len / 254));
}

export class T64Source extends ListSource {
  constructor(bytes: Uint8Array) {
    if (bytes.length < 0x60 || !new TextDecoder().decode(bytes.subarray(0, 3)).startsWith('C64')) {
      throw new DiskError(74, 'not a T64 image');
    }
    const used = bytes[0x24]! | (bytes[0x25]! << 8);
    const max = bytes[0x22]! | (bytes[0x23]! << 8);
    const count = Math.min(Math.max(used, 0), Math.max(max, used), 1000);
    const files: Array<{ file: SourceFile; data: Uint8Array }> = [];
    for (let i = 0; i < count; i++) {
      const e = 0x40 + i * 32;
      if (e + 32 > bytes.length || bytes[e] === 0) continue;
      const start = bytes[e + 2]! | (bytes[e + 3]! << 8);
      const end = bytes[e + 4]! | (bytes[e + 5]! << 8);
      const off = bytes[e + 8]! | (bytes[e + 9]! << 8) | (bytes[e + 10]! << 16) | (bytes[e + 11]! << 24);
      const wanted = Math.max(0, end - start);
      const avail = Math.max(0, Math.min(wanted, bytes.length - off));
      const data = new Uint8Array(2 + avail);
      data[0] = start & 0xff;
      data[1] = start >> 8;
      data.set(bytes.subarray(off, off + avail), 2);
      files.push({
        file: {
          name: trimSpaces(bytes.subarray(e + 16, e + 32)),
          type: 2,
          blocks: blocksOf(data.length),
          closed: true,
          locked: false,
        },
        data,
      });
    }
    super(files, 'T64');
  }
}

export class P00Source extends ListSource {
  constructor(bytes: Uint8Array) {
    if (bytes.length < 26 || new TextDecoder().decode(bytes.subarray(0, 7)) !== 'C64File') {
      throw new DiskError(74, 'not a P00 file');
    }
    const name = trimSpaces(bytes.subarray(8, 24));
    const data = bytes.slice(26);
    super(
      [{ file: { name, type: 2, blocks: blocksOf(data.length), closed: true, locked: false }, data }],
      'P00',
    );
  }
}

export class PrgSource extends ListSource {
  private readonly single: { file: SourceFile; data: Uint8Array };
  constructor(bytes: Uint8Array, name = 'PROGRAM') {
    const file = {
      name: petscii(name).slice(0, 16),
      type: 2,
      blocks: blocksOf(bytes.length),
      closed: true,
      locked: false,
    };
    super([{ file, data: bytes }], name);
    this.single = { file, data: bytes };
  }

  override read(name: Uint8Array) {
    return name.length === 1 && name[0] === 0x2a ? this.single : super.read(name);
  }
}

export function openSource(
  bytes: Uint8Array,
  kind?: 'd64' | 't64' | 'prg' | 'p00',
  name?: string,
): FileSource {
  const sniff = kind ?? detect(bytes);
  switch (sniff) {
    case 'd64':
      return new D64Source(new D64(bytes));
    case 't64':
      return new T64Source(bytes);
    case 'p00':
      return new P00Source(bytes);
    default:
      if (bytes.length < 2) throw new DiskError(74, 'program is shorter than its load address');
      return new PrgSource(bytes, name);
  }
}

function detect(bytes: Uint8Array): 'd64' | 't64' | 'p00' | 'prg' {
  if ([174848, 175531, 196608, 197376].includes(bytes.length)) return 'd64';
  const head = new TextDecoder().decode(bytes.subarray(0, 8));
  if (head.startsWith('C64File')) return 'p00';
  if (head.startsWith('C64S') || head.startsWith('C64 tape') || head.startsWith('C64')) {
    if (bytes.length >= 0x60) return 't64';
  }
  if (bytes.length >= 2) return 'prg';
  throw new DiskError(74, 'unrecognised disk or program image');
}

const TYPE_NAMES = ['DEL', 'SEQ', 'PRG', 'USR', 'REL'];

/**
 * The directory as a BASIC program (what `LOAD"$",8` returns): load address $0401, links computed
 * for that address (BASIC re-links after a relocated LOAD).
 */
export function directoryProgram(src: FileSource, pattern?: Uint8Array): Uint8Array {
  const out: number[] = [0x01, 0x04];
  let addr = 0x0401;
  const line = (num: number, text: number[]) => {
    const len = 2 + 2 + text.length + 1;
    addr += len;
    out.push(addr & 0xff, addr >> 8, num & 0xff, num >> 8, ...text, 0);
  };
  const { name, id } = src.label();
  const head: number[] = [0x12, 0x22];
  for (let i = 0; i < 16; i++) head.push(i < name.length ? name[i]! : 0x20);
  head.push(0x22, 0x20, id[0] ?? 0x30, id[1] ?? 0x30, 0x20, 0x32, 0x41);
  line(0, head);
  for (const f of src.list()) {
    if (pattern && pattern.length > 0 && !matchName(pattern, f.name)) continue;
    const text: number[] = [];
    const pad = f.blocks < 10 ? 3 : f.blocks < 100 ? 2 : f.blocks < 1000 ? 1 : 0;
    for (let i = 0; i < pad; i++) text.push(0x20);
    text.push(0x22, ...f.name, 0x22);
    for (let i = f.name.length; i < 17; i++) text.push(0x20);
    text.push(...new TextEncoder().encode(TYPE_NAMES[f.type] ?? 'PRG'));
    if (!f.closed) text.push(0x2a);
    else if (f.locked) text.push(0x3c);
    else text.push(0x20);
    while (text.length < 25) text.push(0x20);
    line(f.blocks, text);
  }
  const tail = [...new TextEncoder().encode('BLOCKS FREE.'), ...new Array<number>(13).fill(0x20)];
  line(src.blocksFree(), tail);
  out.push(0, 0);
  return Uint8Array.from(out);
}

export function parseFilename(raw: Uint8Array): ParsedName {

    replace = true;
    b = b.slice(1);
  }

    else parts[parts.length - 1]!.push(c);

  let type: number | null = null;
  let mode: 'r' | 'w' | 'a' | null = null;
  for (const p of parts.slice(1)) {
    const c = p[0];
    if (c === undefined) continue;
    if (c === 0x52 && type === null && p.length === 1 && mode === null) mode = 'r';
    else if (c === 0x57 && p.length === 1) mode = 'w';
    else if (c === 0x41 && p.length === 1) mode = 'a';
    else if (TYPE_LETTERS[c] !== undefined && type === null) type = TYPE_LETTERS[c]!;
  }
  const name = Uint8Array.from(parts[0]!);
  return { name, pattern: name, replace, dir, type, mode };

      );
      this.setStatus(code);

      const cmd = Uint8Array.from(this.pendingCommand);
      this.pendingCommand = [];
      this.command(cmd);
    }
  }

    for (const c of [...this.channels.keys()]) this.closeChannel(c);
    this.pendingCommand = [];
  }

    const needsWrite = 'SNVRC'.includes(letter);
    if (needsWrite && !(this.source instanceof D64Source)) return this.setStatus(26);

          const pat = petscii(part.replace(/^\d*:?/, ''));
          for (const e of image!.entries()) {
            if (matchName(pat, e.name) && !e.locked) {
              image!.scratch(e);
              count++;
            }
          }
        }
        return this.setStatus(1, count, 0);
      }
      case 'N': {
        const [name = '', id] = arg.split(',');
        image!.format(name, id);
        return this.setStatus(0);
      }
      case 'V':
        image!.validate();
        return this.setStatus(0);
      case 'I':
        return this.setStatus(0);
      case 'R': {
        const [to = '', from = ''] = arg.split('=');
        const entry = image!.entries().find((e) => matchName(petscii(from), e.name));
        if (!entry) return this.setStatus(62);
        if (image!.entries().some((e) => matchName(petscii(to), e.name))) return this.setStatus(63);
        image!.rename(entry, petscii(to));
        return this.setStatus(0);

      case 'C': {

        if (!entry) return this.setStatus(62);

        const code = this.store(
          { name: petscii(to), pattern: petscii(to), replace: false, dir: false, type: entry.type, mode: 'w' },
          entry.type,
          data,
        );
        return this.setStatus(code);
      }
      default:
        text = '';
        return this.setStatus(31);
    }
  }
}

**Interfaces:**

- Produces:
  - `class KernalTraps { constructor(host: TrapHost); readonly map: Uint8Array; refresh(): void; handle(pc: number): boolean }` where `interface TrapHost { cpu: MOS6502; ram: Uint8Array; peek(addr: number): number; drive(device: number): DiskDrive | null }`.
  - `C64.drive(device: number): ImageDrive` (8-11, created lazily), `C64` instruction loop consults `traps.map[pc]` in `runFrame()` and `runCycles()`, and `traps.refresh()` runs after each `runFrame()` and on `mount`.

**Calling conventions the handlers implement** (standard KERNAL; all addresses are RAM/zero page):

| Vector | Target role | Inputs | Result when handled |
|---|---|---|---|
| `$0330` LOAD | load/verify | `A`=0 load, 1 verify; `X/Y` load address; `$B7` name length, `$BB/$BC` name pointer, `$B9` secondary address, `$BA` device, `$9D` bit 7 = direct mode | carry clear, `X/Y` = end address, `$AE/$AF` = end, `$90` = `$40`; or carry set and `A` = KERNAL error (4 not found, 8 missing name) |
| `$0332` SAVE | save | `A` = zero-page address of the start pointer; `X/Y` end (exclusive); name/device as above | carry clear; carry set + `A` on error (DOS error mapped to 5 device not present only when absent) |
| `$031A` OPEN | open | `$B8` logical file, `$B9` SA, `$BA` device, name as above | carry clear; error 2 when the logical file is already ours |
| `$031C` CLOSE | close | `A` = logical file | carry clear |
| `$031E` CHKIN | input channel | `X` = logical file | carry clear; `$99` = device |
| `$0320` CKOUT | output channel | `X` = logical file | carry clear; `$9A` = device |
| `$0322` CLRCHN | reset channels | - | `$99 = 0`, `$9A = 3`; carry clear |
| `$0324` CHRIN / `$032A` GETIN | read byte | - (current input is ours) | `A` = byte, carry clear, `$90` = `$40` at end of file |
| `$0326` CHROUT | write byte | `A` = byte (current output is ours) | carry clear, `A` unchanged |
| `$032C` CLALL | close all | - | closes ours, then the ROM routine runs |

A handler returns `true` when it consumed the call (the PC is set to the caller's return address and the stack popped); `false` lets the ROM routine execute unchanged.

export interface TrapHost {
  cpu: MOS6502;
  ram: Uint8Array;
  /** CPU-visible byte (banking applied, no side effects). */

}

const enum Kind {
  None = 0,
  Open,
  Close,
  Chkin,
  Ckout,
  Clrchn,
  Chrin,
  Chrout,
  Getin,
  Clall,
  Load,
  Save,
  Resume,
}

const VECTORS: Array<[number, Kind]> = [
  [0x031a, Kind.Open],
  [0x031c, Kind.Close],
  [0x031e, Kind.Chkin],
  [0x0320, Kind.Ckout],
  [0x0322, Kind.Clrchn],
  [0x0324, Kind.Chrin],
  [0x0326, Kind.Chrout],
  [0x032a, Kind.Getin],
  [0x032c, Kind.Clall],
  [0x0330, Kind.Load],
  [0x0332, Kind.Save],
];

/** Message stub lives in the cassette buffer; it prints through CHROUT and jumps to RESUME. */
const STUB = 0x0350;
const RESUME = 0x03f8;
const MSG = STUB + 16;
const C = 0x01;

interface Saved {
  a: number;
  x: number;
  y: number;
}

export class KernalTraps {
  readonly map = new Uint8Array(0x10000);
  private readonly files = new Map<number, { device: number; sa: number }>();
  private currentIn = -1;
  private currentOut = -1;
  private pending: (() => void) | null = null;

  constructor(private readonly host: TrapHost) {}

  /** Re-read the KERNAL vectors; call after boot, reset and when a drive is mounted. */
  refresh(): void {
    this.map.fill(Kind.None);
    const ram = this.host.ram;
    for (const [vec, kind] of VECTORS) {
      const target = ram[vec]! | (ram[vec + 1]! << 8);
      if (target >= 0x0200) this.map[target] = kind;
    }
    this.map[RESUME] = Kind.Resume;
  }

  /** Returns true when the instruction at `pc` was replaced by a handler. */
  handle(pc: number): boolean {
    switch (this.map[pc] as Kind) {
      case Kind.Load:
        return this.load();
      case Kind.Save:
        return this.save();
      case Kind.Open:
        return this.open();
      case Kind.Close:
        return this.close();
      case Kind.Chkin:
        return this.chk(true);
      case Kind.Ckout:
        return this.chk(false);
      case Kind.Clrchn:
        return this.clrchn();
      case Kind.Chrin:
      case Kind.Getin:
        return this.chrin();
      case Kind.Chrout:
        return this.chrout();
      case Kind.Clall:
        this.closeAll();
        return false;
      case Kind.Resume: {
        const p = this.pending;
        this.pending = null;
        if (p) p();
        return true;
      }
      default:
        return false;
    }
  }

  // ---- helpers -----------------------------------------------------------------

  private ram(): Uint8Array {
    return this.host.ram;
  }

  private nameBytes(): Uint8Array {
    const ram = this.ram();
    const len = ram[0xb7]!;
    const ptr = ram[0xbb]! | (ram[0xbc]! << 8);
    const out = new Uint8Array(len);
    for (let i = 0; i < len; i++) out[i] = this.host.peek((ptr + i) & 0xffff);
    return out;
  }

  /** Pop the return address and continue after the caller's JSR with the given carry. */
  private ret(carry: boolean, regs?: Partial<Saved>): true {
    const cpu = this.host.cpu;
    const r = cpu.regs;
    const ram = this.ram();
    const lo = ram[0x100 + ((r.sp + 1) & 0xff)]!;
    const hi = ram[0x100 + ((r.sp + 2) & 0xff)]!;
    r.sp = (r.sp + 2) & 0xff;
    r.pc = (((hi << 8) | lo) + 1) & 0xffff;
    r.p = carry ? r.p | C : r.p & ~C;
    if (regs?.a !== undefined) r.a = regs.a & 0xff;
    if (regs?.x !== undefined) r.x = regs.x & 0xff;
    if (regs?.y !== undefined) r.y = regs.y & 0xff;
    return true;
  }

  /** Print through the ROM's CHROUT by running a stub, then run `then`. */
  private print(text: string, then: () => void): void {
    const ram = this.ram();
    const stub = [
      0xa2, 0x00, // LDX #0
      0xbd, MSG & 0xff, MSG >> 8, // LDA MSG,X
      0xf0, 0x06, // BEQ done
      0x20, 0xd2, 0xff, // JSR $FFD2
      0xe8, // INX
      0xd0, 0xf5, // BNE loop
      0x4c, RESUME & 0xff, RESUME >> 8, // JMP RESUME
    ];
    stub.forEach((b, i) => (ram[STUB + i] = b));
    const bytes = [...text].map((c) => c.charCodeAt(0) & 0xff).slice(0, 60);
    bytes.forEach((b, i) => (ram[MSG + i] = b));
    ram[MSG + bytes.length] = 0;
    this.pending = then;
    this.host.cpu.regs.pc = STUB;
  }

  private ourFile(lf: number): { device: number; sa: number; drive: DiskDrive } | null {
    const f = this.files.get(lf);
    if (!f) return null;
    const drive = this.host.drive(f.device);
    return drive ? { ...f, drive } : null;
  }

  // ---- LOAD / SAVE ---------------------------------------------------------------

  private load(): boolean {
    const ram = this.ram();
    const drive = this.host.drive(ram[0xba]!);
    if (!drive) return false;
    const r = this.host.cpu.regs;
    const verify = r.a === 1;
    const sa = ram[0xb9]!;
    const x = r.x;
    const y = r.y;
    if (ram[0xb7] === 0) return this.ret(true, { a: 8 });
    const name = this.nameBytes();
    const direct = (ram[0x9d]! & 0x80) !== 0;
    const finish = () => {
      const res = drive.loadFile(name);
      if ('error' in res) return this.ret(true, { a: res.error === 74 ? 5 : 4 });
      const data = res.data;
      if (data.length < 2) return this.ret(true, { a: 4 });
      const start = sa === 0 ? x | (y << 8) : data[0]! | (data[1]! << 8);
      let status = 0x40;
      let addr = start;
      for (let i = 2; i < data.length; i++, addr = (addr + 1) & 0xffff) {
        if (verify) {
          if (this.host.peek(addr) !== data[i]) status |= 0x10;
        } else {
          ram[addr] = data[i]!;
        }
      }
      ram[0x90] = status;
      ram[0xae] = addr & 0xff;
      ram[0xaf] = addr >> 8;
      return this.ret(false, { x: addr & 0xff, y: addr >> 8 });
    };
    if (!direct) return finish();
    const label = String.fromCharCode(...name);
    this.print(`SEARCHING FOR ${label}\r`, () => {
      const probe = drive.loadFile(name);
      if ('error' in probe) {
        this.ret(true, { a: probe.error === 74 ? 5 : 4 });
        return;
      }
      this.print(verify ? 'VERIFYING\r' : 'LOADING\r', () => void finish());
    });
    return true;
  }

  private save(): boolean {
    const ram = this.ram();
    const drive = this.host.drive(ram[0xba]!);
    if (!drive) return false;
    const r = this.host.cpu.regs;
    const zp = r.a;
    const start = ram[zp]! | (ram[(zp + 1) & 0xff]! << 8);
    const end = r.x | (r.y << 8);
    if (ram[0xb7] === 0) return this.ret(true, { a: 8 });
    const name = this.nameBytes();
    const direct = (ram[0x9d]! & 0x80) !== 0;
    const finish = () => {
      const len = Math.max(0, (end - start) & 0xffff);
      const data = new Uint8Array(2 + len);
      data[0] = start & 0xff;
      data[1] = start >> 8;
      for (let i = 0; i < len; i++) data[2 + i] = this.host.peek((start + i) & 0xffff);
      const code = drive.saveFile(name, data);
      return code === 0 ? this.ret(false) : this.ret(true, { a: code === 74 ? 5 : 4 });
    };
    if (!direct) return finish();
    this.print(`SAVING ${String.fromCharCode(...name)}\r`, () => void finish());
    return true;
  }

  // ---- channels ----------------------------------------------------------------------

  private open(): boolean {
    const ram = this.ram();
    const device = ram[0xba]!;
    const drive = this.host.drive(device);
    if (!drive) return false;
    const lf = ram[0xb8]!;
    const sa = ram[0xb9]!;
    if (this.files.has(lf)) return this.ret(true, { a: 2 });
    this.files.set(lf, { device, sa });
    drive.open(sa & 0x0f, this.nameBytes());
    return this.ret(false);
  }

  private close(): boolean {
    const lf = this.host.cpu.regs.a;
    const f = this.ourFile(lf);
    if (!f) return false;
    f.drive.close(f.sa & 0x0f);
    this.files.delete(lf);
    if (this.currentIn === lf) this.currentIn = -1;
    if (this.currentOut === lf) this.currentOut = -1;
    return this.ret(false);
  }

  private chk(input: boolean): boolean {
    const lf = this.host.cpu.regs.x;
    const f = this.ourFile(lf);
    if (!f) return false;
    const ram = this.ram();
    if (input) {
      this.currentIn = lf;
      ram[0x99] = f.device;
    } else {
      this.currentOut = lf;
      ram[0x9a] = f.device;
    }
    return this.ret(false);
  }

  private clrchn(): boolean {
    if (this.currentIn < 0 && this.currentOut < 0) return false;
    const ram = this.ram();
    if (this.currentIn >= 0) ram[0x99] = 0;
    if (this.currentOut >= 0) ram[0x9a] = 3;
    this.currentIn = this.currentOut = -1;
    return this.ret(false);
  }

  private closeAll(): void {
    for (const [lf, f] of this.files) {
      this.host.drive(f.device)?.close(f.sa & 0x0f);
      this.files.delete(lf);
    }
    this.currentIn = this.currentOut = -1;
  }

  private chrin(): boolean {
    if (this.currentIn < 0) return false;
    const f = this.ourFile(this.currentIn);
    if (!f || this.ram()[0x99] !== f.device) return false;
    const got = f.drive.read(f.sa & 0x0f);
    this.ram()[0x90] = got.eof ? 0x40 : 0;
    return this.ret(false, { a: got.byte });
  }

  private chrout(): boolean {
    if (this.currentOut < 0) return false;
    const f = this.ourFile(this.currentOut);
    if (!f || this.ram()[0x9a] !== f.device) return false;
    f.drive.write(f.sa & 0x0f, this.host.cpu.regs.a);
    return this.ret(false);
  }

  /** Forget open files (machine reset). */
  reset(): void {
    this.files.clear();
    this.currentIn = this.currentOut = -1;
    this.pending = null;
  }
}

const HEADER = 'C64-TAPE-RAW';

/** Parse a TAP file into pulse lengths (CPU cycles between falling edges). */
export function parseTap(bytes: Uint8Array): Uint32Array {
  if (bytes.length < 20 || new TextDecoder().decode(bytes.subarray(0, 12)) !== HEADER) {
    throw new Error('not a TAP file (bad header)');
  }
  const version = bytes[12]!;
  const size = bytes[16]! | (bytes[17]! << 8) | (bytes[18]! << 16) | (bytes[19]! << 24);
  const end = Math.min(bytes.length, 20 + (size >>> 0));
  const out: number[] = [];
  let half = -1;
  for (let i = 20; i < end; i++) {
    const b = bytes[i]!;
    let cycles: number;
    if (b !== 0) {
      cycles = b * 8;
    } else if (version === 0) {
      cycles = 256 * 8;
    } else {
      if (i + 3 >= end + 0 && i + 3 > bytes.length - 1 + 0 && i + 3 >= bytes.length) {
        throw new Error('truncated TAP data (cycle count)');
      }
      if (i + 3 >= end) throw new Error('truncated TAP data (cycle count)');
      cycles = bytes[i + 1]! | (bytes[i + 2]! << 8) | (bytes[i + 3]! << 16);
      i += 3;
    }
    if (version === 2) {
      if (half < 0) half = cycles;
      else {
        out.push(half + cycles);
        half = -1;
      }
    } else {
      out.push(cycles);
    }
  }
  return Uint32Array.from(out);
}

interface Host {
  motorOn(): boolean;
  pulse(): void;
}

export class Datasette {
  private pulses = new Uint32Array(0);
  private index = 0;
  private countdown = 0;
  private pressed = false;

  constructor(private readonly host: Host) {}

  insert(bytes: Uint8Array): void {
    this.pulses = parseTap(bytes);
    this.rewind();
    this.pressed = false;
  }

  eject(): void {
    this.pulses = new Uint32Array(0);
    this.rewind();
    this.pressed = false;
  }

  play(): void {
    if (this.pulses.length > 0 && this.index < this.pulses.length) this.pressed = true;
  }

  stop(): void {
    this.pressed = false;
  }

  rewind(): void {
    this.index = 0;
    this.countdown = this.pulses.length > 0 ? this.pulses[0]! : 0;
  }

  /** The PLAY key is down (the cassette sense line reads low). */
  get sense(): boolean {
    return this.pressed;
  }

  get playing(): boolean {
    return this.pressed;
  }

  /** Playing and the motor is on: the tape is moving. */
  get running(): boolean {
    return this.pressed && this.host.motorOn();
  }

  get position(): number {
    return this.index;
  }

  get length(): number {
    return this.pulses.length;
  }

  tick(): void {
    if (!this.pressed || !this.host.motorOn()) return;
    if (--this.countdown > 0) return;
    this.host.pulse();
    this.index++;
    if (this.index >= this.pulses.length) {
      this.pressed = false; // end of tape: the key pops up
      return;
    }
    this.countdown = Math.max(1, this.pulses[this.index]!);
  }
}

- [ ] **Step 2: Write the KERNAL tape encoder**

`test/conformance/tap-encoder.ts` (documented standard format: leader of short pulses, a countdown/sync sequence, then bytes as marker + 8 data bits + parity pulse pairs, header block then data block, each block twice):
```ts
const SHORT = 0x30;
const MEDIUM = 0x42;
const LONG = 0x56;

function byteInto(out: number[], value: number): void {
  out.push(LONG, MEDIUM); // new-byte marker
  let parity = 1;
  for (let i = 0; i < 8; i++) {
    const bit = (value >> i) & 1;
    parity ^= bit;
    if (bit) out.push(MEDIUM, SHORT);
    else out.push(SHORT, MEDIUM);
  }
  if (parity) out.push(MEDIUM, SHORT);
  else out.push(SHORT, MEDIUM);
}

function block(out: number[], payload: number[], repeat: boolean): void {
  for (let i = 0; i < 0x6a00 / 4; i++) out.push(SHORT); // leader
  out.push(LONG, MEDIUM); // sync start
  const count = repeat ? [0x09, 0x08, 0x07, 0x06, 0x05, 0x04, 0x03, 0x02, 0x01] : [0x89, 0x88, 0x87, 0x86, 0x85, 0x84, 0x83, 0x82, 0x81];
  for (const c of count) byteInto(out, c);
  let checksum = 0;
  for (const b of payload) {
    byteInto(out, b);
    checksum ^= b;
  }
  byteInto(out, checksum);
  out.push(LONG, SHORT); // end of data
}

/** A standard KERNAL tape: header block (type 3, absolute) and data block, each recorded twice. */
export function encodeKernalTap(name: string, program: Uint8Array): Uint8Array {
  const start = program[0]! | (program[1]! << 8);
  const data = Array.from(program.subarray(2));
  const end = start + data.length;
  const header = [0x03, start & 0xff, start >> 8, end & 0xff, end >> 8];
  const padded = name.toUpperCase().padEnd(16, ' ').slice(0, 16);
  for (const c of padded) header.push(c.charCodeAt(0));
  while (header.length < 192) header.push(0x20);
  const pulses: number[] = [];
  block(pulses, header, false);
  block(pulses, header, true);
  for (let i = 0; i < 80; i++) pulses.push(SHORT); // gap
  block(pulses, data, false);
  block(pulses, data, true);
  const out = new Uint8Array(20 + pulses.length);
  out.set(new TextEncoder().encode('C64-TAPE-RAW'), 0);
  out[12] = 0;
  out[16] = pulses.length & 0xff;
  out[17] = (pulses.length >> 8) & 0xff;
  out[18] = (pulses.length >> 16) & 0xff;
  out.set(pulses, 20);
  return out;
}
```
