## Review Focus

- A write to a SID register while a note plays (gate, frequency, control) must never produce NaN/Infinity samples; the output stays within [-1, 1].
- Switching the model at runtime (`sid.model = '8580'`) mid-note must not click violently or reset voice state.
- `drainSamples` with a too-small buffer must return only what fits and keep the rest; calling it never throws or reads stale data after a machine `reset()`.
- A machine that never drains (headless runs, `runFrame` loops) must not grow memory without bound.
- `start({ audio: true })` in a browser without `AudioWorklet` or before a user gesture must degrade to silence with a console notice, not throw, and must not break video or keyboard.
- Sustained noise waveform combined with other waveforms must not lock the LFSR at zero forever (the write-back can clear it); the model must keep it alive.

```ts
/** Rate counter periods (cycles) for the 16 attack/decay/release rate values. */
export const RATE_PERIOD = [
  9, 32, 63, 95, 149, 220, 267, 313, 392, 977, 1954, 3126, 3907, 11720, 19532, 31251,
] as const;

const ATTACK = 0;
const DECAY = 1;
const RELEASE = 2;

/** One SID envelope generator: a 15-bit rate counter driving an 8-bit counter with an exponential stage. */
export class Envelope {
  /** The 8-bit envelope value (what ENV3 reads). */
  counter = 0;

  private state = RELEASE;
  private gate = false;
  private rateCounter = 0;
  private ratePeriod: number = RATE_PERIOD[0];
  private expCounter = 0;
  private expPeriod = 1;
  private holdZero = true;
  private attack = 0;
  private decay = 0;
  private sustain = 0; // 4-bit
  private release = 0;

  reset(): void {
    this.counter = 0;
    this.state = RELEASE;
    this.gate = false;
    this.rateCounter = 0;
    this.expCounter = 0;
    this.expPeriod = 1;
    this.holdZero = true;
    this.attack = this.decay = this.sustain = this.release = 0;
    this.ratePeriod = RATE_PERIOD[0];
  }

  setGate(on: boolean): void {
    if (on === this.gate) return;
    this.gate = on;
    if (on) {
      this.state = ATTACK;
      this.ratePeriod = RATE_PERIOD[this.attack]!;
      this.holdZero = false;
    } else {
      this.state = RELEASE;
      this.ratePeriod = RATE_PERIOD[this.release]!;
    }
  }

  setAttackDecay(value: number): void {
    this.attack = (value >> 4) & 0x0f;
    this.decay = value & 0x0f;
    if (this.state === ATTACK) this.ratePeriod = RATE_PERIOD[this.attack]!;
    else if (this.state === DECAY) this.ratePeriod = RATE_PERIOD[this.decay]!;
  }

  setSustainRelease(value: number): void {
    this.sustain = (value >> 4) & 0x0f;
    this.release = value & 0x0f;
    if (this.state === RELEASE) this.ratePeriod = RATE_PERIOD[this.release]!;
  }

  /** One cycle. The rate counter is 15 bits and is only cleared when it matches the period. */
  tick(): void {
    this.rateCounter = (this.rateCounter + 1) & 0x7fff;
    if (this.rateCounter !== this.ratePeriod) return;
    this.rateCounter = 0;
    if (this.state !== ATTACK && ++this.expCounter !== this.expPeriod) return;
    this.expCounter = 0;
    if (this.holdZero) return;
    switch (this.state) {
      case ATTACK:
        this.counter = (this.counter + 1) & 0xff;
        if (this.counter === 0xff) {
          this.state = DECAY;
          this.ratePeriod = RATE_PERIOD[this.decay]!;
        }
        break;
      case DECAY:
        if (this.counter !== this.sustain * 17) this.counter = (this.counter - 1) & 0xff;
        break;
      default:
        this.counter = (this.counter - 1) & 0xff;
    }
    switch (this.counter) {
      case 0xff:
        this.expPeriod = 1;
        break;
      case 0x5d:
        this.expPeriod = 2;
        break;
      case 0x36:
        this.expPeriod = 4;
        break;
      case 0x1a:
        this.expPeriod = 8;
        break;
      case 0x0e:
        this.expPeriod = 16;
        break;
      case 0x06:
        this.expPeriod = 30;
        break;
      case 0x00:
        this.expPeriod = 1;
        this.holdZero = true;
        break;
    }
  }
}
```

```ts
export interface SidModelParams {
  name: '6581' | '8580';
  /** Waveform "zero" level (12-bit): the value that produces no output after the envelope DAC. */
  zero: number;
  /** Leakage applied to the AND of combined waveforms (analytical approximation, not measured tables). */
  combine(and: number, selection: number): number;
  /** Filter cutoff in Hz for an 11-bit FC value. */
  cutoffHz(fc: number): number;
  /** Filter damping Q for a 4-bit resonance value. */
  q(resonance: number): number;
}

export const MODEL_8580: SidModelParams = {
  name: '8580',
  zero: 0x800,
  combine: (and) => and,
  cutoffHz: (fc) => 30 + (12000 * fc) / 2047,
  q: (r) => 0.707 + (r / 15) * 6,
};

export const MODEL_6581: SidModelParams = {
  name: '6581',
  zero: 0x380,
  // Weak output stages: a combined bit only survives if a neighbouring bit is also set.
  combine: (and) => and & (((and << 1) | (and >> 1) | 1) & 0xfff),
  cutoffHz: (fc) => 200 * Math.pow(60, fc / 2047),
  q: (r) => 0.707 + (r / 15) * 8,
};

export function modelByName(name: '6581' | '8580'): SidModelParams {
  return name === '8580' ? MODEL_8580 : MODEL_6581;
}
```

```ts
import type { SidModelParams } from './sid-model';

const NOISE_RESET = 0x7ffff8;
/** Cycles the TEST bit must be held before the LFSR is reset. */
const TEST_LFSR_DELAY = 0x8000;
/** Cycles after which a floating (no waveform selected) output has decayed to zero. */
const FLOAT_DECAY_CYCLES = 2_500_000;

/** One SID voice: 24-bit phase accumulator, noise LFSR and the waveform generators. */
export class Voice {
  acc = 0x555555;
  lfsr = NOISE_RESET;
  freq = 0;
  pw = 0;
  control = 0;
  /** The neighbouring voice that drives sync and ring modulation (set by `Sid`). */
  source: Voice = this;
  /** True for the cycle in which the accumulator MSB went from 0 to 1. */
  msbRising = false;

  private testCycles = 0;
  private floating = 0;
  private floatAge = 0;

  reset(): void {
    this.acc = 0x555555; // power-up value; real reset leaves it unchanged, which the harness mirrors
    this.lfsr = NOISE_RESET;
    this.freq = 0;
    this.pw = 0;
    this.control = 0;
    this.msbRising = false;
    this.testCycles = 0;
    this.floating = 0;
    this.floatAge = 0;
  }

  /** Advance the oscillator by one cycle (before sync is applied). */
  advance(): void {
    const old = this.acc;
    if (this.control & 0x08) {
      this.acc = 0;
      this.msbRising = false;
      if (++this.testCycles >= TEST_LFSR_DELAY) this.lfsr = NOISE_RESET;
    } else {
      this.testCycles = 0;
      this.acc = (old + this.freq) & 0xffffff;
      this.msbRising = (old & 0x800000) === 0 && (this.acc & 0x800000) !== 0;
      if ((old & 0x080000) === 0 && (this.acc & 0x080000) !== 0) this.clockNoise();
    }
    if (this.floatAge < FLOAT_DECAY_CYCLES) this.floatAge++;
  }

  /** Hard sync: if enabled, a rising MSB of the source voice resets this accumulator. */
  applySync(): void {
    if ((this.control & 0x02) !== 0 && this.source.msbRising) this.acc = 0;
  }

  private clockNoise(): void {
    const bit0 = ((this.lfsr >> 22) ^ (this.lfsr >> 17)) & 1;
    this.lfsr = ((this.lfsr << 1) | bit0) & 0x7fffff;
  }

  private noiseOut(): number {
    const l = this.lfsr;
    return (
      (((l >> 20) & 1) << 11) |
      (((l >> 18) & 1) << 10) |
      (((l >> 14) & 1) << 9) |
      (((l >> 11) & 1) << 8) |
      (((l >> 9) & 1) << 7) |
      (((l >> 5) & 1) << 6) |
      (((l >> 2) & 1) << 5) |
      ((l & 1) << 4)
    );
  }

  /** 12-bit waveform output for the current accumulator and control bits. */
  wave(model: SidModelParams): number {
    const sel = this.control & 0xf0;
    if (sel === 0) {
      if (this.floatAge >= FLOAT_DECAY_CYCLES) this.floating = 0;
      return this.floating;
    }
    const acc = this.acc;
    let out = 0xfff;
    let waves = 0;
    if (sel & 0x10) {
      const ring = (this.control & 0x04) !== 0;
      const srcMsb = (this.source.acc >> 23) & 1;
      const invert = (sel & 0x20) === 0 && ((ring && srcMsb === 0 ? 1 : 0) ^ ((acc >> 23) & 1)) === 1;
      out &= ((invert ? ~acc & 0xffffff : acc) >> 11) & 0xfff;
      waves++;
    }
    if (sel & 0x20) {
      out &= (acc >> 12) & 0xfff;
      waves++;
    }
    if (sel & 0x40) {
      out &= this.control & 0x08 || acc >> 12 >= this.pw ? 0xfff : 0;
      waves++;
    }
    if (waves > 1) out = model.combine(out, sel);
    if (sel & 0x80) {
      if (waves === 0) {
        out = this.noiseOut();
      } else {
        // Noise combined with another waveform pulls the LFSR bits towards zero (write-back).
        const noise = this.noiseOut();
        out &= noise;
        this.writeBack(out);
        out &= noise;
      }
    }
    this.floating = out;
    this.floatAge = 0;
    return out;
  }

  private writeBack(out: number): void {
    const taps = [20, 18, 14, 11, 9, 5, 2, 0];
    let l = this.lfsr;
    for (let i = 0; i < 8; i++) {
      if (((out >> (11 - i)) & 1) === 0) l &= ~(1 << taps[i]!);
    }
    // A dead register would stay dead: the real chip recovers through the test bit; keep it alive.
    this.lfsr = l === 0 ? NOISE_RESET : l;
  }
}
```

```ts
/** Fixed-capacity ring of output samples; the oldest samples are dropped on overflow. */
export class SampleRing {
  private readonly buf: Float32Array;
  private head = 0;
  private size = 0;

  constructor(readonly capacity: number) {
    this.buf = new Float32Array(capacity);
  }

  get length(): number {
    return this.size;
  }

  push(v: number): void {
    const tail = (this.head + this.size) % this.capacity;
    this.buf[tail] = v;
    if (this.size < this.capacity) this.size++;
    else this.head = (this.head + 1) % this.capacity;
  }

  drain(out: Float32Array): number {
    const n = Math.min(out.length, this.size);
    for (let i = 0; i < n; i++) out[i] = this.buf[(this.head + i) % this.capacity]!;
    this.head = (this.head + n) % this.capacity;
    this.size -= n;
    return n;
  }

  clear(): void {
    this.head = 0;
    this.size = 0;
  }
}
```

```ts
import type { IoChip } from './io';
import { Envelope } from './sid-envelope';
import { modelByName, type SidModelParams } from './sid-model';
import { SampleRing } from './sid-output';
import { Voice } from './sid-voice';

export const SID_CLOCK = 985_248;
const BUS_DECAY_CYCLES = 8192;

export interface SidOptions {
  model?: '6581' | '8580';
  sampleRate?: number;
}

export class Sid implements IoChip {
  readonly sampleRate: number;
  readonly voices = [new Voice(), new Voice(), new Voice()];
  readonly envelopes = [new Envelope(), new Envelope(), new Envelope()];
  protected readonly regs = new Uint8Array(0x20);
  protected modelParams: SidModelParams;
  protected readonly ring: SampleRing;
  private busValue = 0;
  private busAge = 0;

  constructor(options: SidOptions = {}) {
    this.sampleRate = options.sampleRate ?? 48000;
    this.modelParams = modelByName(options.model ?? '6581');
    this.ring = new SampleRing(this.sampleRate); // one second
    // Sync / ring mod sources: voice 1 <- voice 3, voice 2 <- voice 1, voice 3 <- voice 2.
    this.voices[0]!.source = this.voices[2]!;
    this.voices[1]!.source = this.voices[0]!;
    this.voices[2]!.source = this.voices[1]!;
  }

  get model(): '6581' | '8580' {
    return this.modelParams.name;
  }

  set model(name: '6581' | '8580') {
    this.modelParams = modelByName(name);
  }

  reset(): void {
    this.regs.fill(0);
    for (const e of this.envelopes) e.reset();
    for (const v of this.voices) v.reset();
    this.busValue = 0;
    this.busAge = 0;
    this.ring.clear();
  }

  read(reg: number): number {
    const v = this.peek(reg);
    return v;
  }

  peek(reg: number): number {
    reg &= 0x1f;
    switch (reg) {
      case 0x19:
      case 0x1a:
        return 0xff;
      case 0x1b:
        return this.voices[2]!.wave(this.modelParams) >> 4;
      case 0x1c:
        return this.envelopes[2]!.counter;
      default:
        return this.busValue;
    }
  }

  write(reg: number, value: number): void {
    reg &= 0x1f;
    value &= 0xff;
    this.busValue = value;
    this.busAge = 0;
    if (reg >= 0x19) {
      this.regs[reg] = value;
      return;
    }
    this.regs[reg] = value;
    if (reg < 0x15) {
      const n = (reg / 7) | 0;
      const r = reg - n * 7;
      const v = this.voices[n]!;
      const e = this.envelopes[n]!;
      switch (r) {
        case 0:
          v.freq = (v.freq & 0xff00) | value;
          break;
        case 1:
          v.freq = (v.freq & 0x00ff) | (value << 8);
          break;
        case 2:
          v.pw = (v.pw & 0x0f00) | value;
          break;
        case 3:
          v.pw = (v.pw & 0x00ff) | ((value & 0x0f) << 8);
          break;
        case 4:
          v.control = value;
          e.setGate((value & 1) !== 0);
          break;
        case 5:
          e.setAttackDecay(value);
          break;
        default:
          e.setSustainRelease(value);
      }
    }
  }

  /** One CPU cycle. */
  tick(): void {
    const [v0, v1, v2] = this.voices as [Voice, Voice, Voice];
    v0.advance();
    v1.advance();
    v2.advance();
    v0.applySync();
    v1.applySync();
    v2.applySync();
    const e = this.envelopes;
    e[0]!.tick();
    e[1]!.tick();
    e[2]!.tick();
    if (this.busAge < BUS_DECAY_CYCLES && ++this.busAge >= BUS_DECAY_CYCLES) this.busValue = 0;
    this.mix();
  }

  /** Overridden in Task 4: produces output samples. */
  protected mix(): void {}

  drainSamples(out: Float32Array): number {
    return this.ring.drain(out);
  }
}
```

```ts
    // Sample the voice outputs every 4th cycle and box-average them per output sample.
    if ((this.subCycle = (this.subCycle + 1) & 3) === 0) {
      const m = this.modelParams;
      const filterRoute = this.regs[0x17]!;
      let dry = 0;
      let filt = 0;
      for (let n = 0; n < 3; n++) {
        if (n === 2 && (this.regs[0x18]! & 0x80) !== 0 && (filterRoute & 0x04) === 0) continue; // voice 3 off
        const w = this.voices[n]!.wave(m);
        const v = ((w - m.zero) / 2048) * (this.envelopes[n]!.counter / 255);
        if ((filterRoute >> n) & 1) filt += v;
        else dry += v;
      }
      this.accDry += dry;
      this.accFilt += filt;
      this.accCount++;
    }
    this.phase += this.sampleRate;
    if (this.phase < SID_CLOCK) return;
    this.phase -= SID_CLOCK;
    const cnt = this.accCount || 1;
    const dry = this.accDry / cnt;
    const filtIn = this.accFilt / cnt;
    this.accDry = this.accFilt = 0;
    this.accCount = 0;

    const mode = this.regs[0x18]!;
    const fc = this.regs[0x15]! & 7 | (this.regs[0x16]! << 3);
    const res = this.regs[0x17]! >> 4;
    const m = this.modelParams;
    const f = Math.min(2 * Math.sin((Math.PI * m.cutoffHz(fc)) / this.sampleRate), 1.2);
    const q = 1 / m.q(res);
    const hp = filtIn - this.lp - q * this.bp;
    this.bp += f * hp;
    this.lp += f * this.bp;
    if (!Number.isFinite(this.lp) || !Number.isFinite(this.bp)) this.lp = this.bp = 0;
    this.bp = Math.max(-8, Math.min(8, this.bp));
    this.lp = Math.max(-8, Math.min(8, this.lp));
    let filtered = 0;
    if (mode & 0x10) filtered += this.lp;
    if (mode & 0x20) filtered += this.bp;
    if (mode & 0x40) filtered += hp;
    const vol = (mode & 0x0f) / 15;
    let s = ((dry + filtered) / 3) * vol;
    // One-pole DC blocker (about 10 Hz): the per-model zero level and volume changes are DC.
    const y = s - this.dcIn + 0.9987 * this.dcOut;
    this.dcIn = s;
    this.dcOut = y;
    s = Math.max(-1, Math.min(1, y));
    this.ring.push(s);
```

```ts
  let out = '';
```

```ts
export interface AudioSink {
  push(samples: Float32Array): void;
  setMuted(muted: boolean): void;
  close(): void;
}

interface SampleSource {
  readonly sampleRate: number;
  drainSamples(out: Float32Array): number;
}

const CHUNK = 4096;

/** Moves the samples the SID produced during the last emulated frame to a sink. */
export class AudioPump {
  private readonly buffer = new Float32Array(CHUNK);
  private muted = false;
  private closed = false;

  constructor(
    private readonly source: SampleSource,
    private readonly sink: AudioSink,
  ) {}

  pump(): number {
    if (this.closed) return 0;
    let total = 0;
    for (;;) {
      const n = this.source.drainSamples(this.buffer);
      if (n === 0) break;
      total += n;
      if (!this.muted) this.sink.push(this.buffer.slice(0, n));
      if (n < this.buffer.length) break;
    }
    return this.muted ? 0 : total;
  }

  setMuted(muted: boolean): void {
    this.muted = muted;
    this.sink.setMuted(muted);
  }

  close(): void {
    if (this.closed) return;
    this.closed = true;
    this.sink.close();
  }
}

/** Worklet source: a ring buffer that plays at a rate nudged towards the target queue fill. */
const WORKLET_SOURCE = `
class F6502Processor extends AudioWorkletProcessor {
  constructor(options) {
    super();
    const o = options.processorOptions;
    this.ratioBase = o.sampleRate / sampleRate;
    this.target = Math.max(256, Math.round(o.sampleRate * o.targetMs / 1000));
    this.buf = new Float32Array(o.sampleRate);
    this.head = 0; this.size = 0; this.pos = 0; this.last = 0; this.muted = false;
    this.port.onmessage = (e) => {
      const m = e.data;
      if (m.muted !== undefined) { this.muted = m.muted; return; }
      const s = m.samples;
      for (let i = 0; i < s.length; i++) {
        const tail = (this.head + this.size) % this.buf.length;
        this.buf[tail] = s[i];
        if (this.size < this.buf.length) this.size++; else this.head = (this.head + 1) % this.buf.length;
      }
    };
  }
  process(inputs, outputs) {
    const out = outputs[0][0];
    const err = (this.size - this.target) / this.target;
    const ratio = this.ratioBase * (1 + Math.max(-0.005, Math.min(0.005, err * 0.01)));
    for (let i = 0; i < out.length; i++) {
      if (this.muted || this.size === 0) { out[i] = this.muted ? 0 : this.last; continue; }
      const idx = (this.head + 0) % this.buf.length;
      this.last = this.buf[idx];
      out[i] = this.last;
      this.pos += ratio;
      while (this.pos >= 1 && this.size > 0) {
        this.pos -= 1; this.head = (this.head + 1) % this.buf.length; this.size--;
      }
    }
    for (let c = 1; c < outputs[0].length; c++) outputs[0][c].set(out);
    return true;
  }
}
registerProcessor('f6502-sid', F6502Processor);
`;

/** Creates the Web Audio sink, or null when AudioWorklet is unavailable. Never throws. */
export async function createWorkletSink(options: {
  sampleRate: number;
  targetMs?: number;
}): Promise<(AudioSink & { resume(): void }) | null> {
  try {
    const Ctx: typeof AudioContext | undefined =
      typeof AudioContext !== 'undefined' ? AudioContext : undefined;
    if (!Ctx || typeof AudioWorkletNode === 'undefined') {
      console.warn('f6502: AudioWorklet is not available; running silent');
      return null;
    }
    const ctx = new Ctx({ sampleRate: options.sampleRate, latencyHint: 'interactive' });
    const url = URL.createObjectURL(new Blob([WORKLET_SOURCE], { type: 'text/javascript' }));
    await ctx.audioWorklet.addModule(url);
    URL.revokeObjectURL(url);
    const node = new AudioWorkletNode(ctx, 'f6502-sid', {
      numberOfInputs: 0,
      outputChannelCount: [2],
      processorOptions: { sampleRate: options.sampleRate, targetMs: options.targetMs ?? 60 },
    });
    node.connect(ctx.destination);
    return {
      push: (samples) => node.port.postMessage({ samples }, [samples.buffer]),
      setMuted: (muted) => node.port.postMessage({ muted }),
      resume: () => void ctx.resume(),
      close: () => {
        node.disconnect();
        void ctx.close();
      },
    };
  } catch (error) {
    console.warn('f6502: audio could not start; running silent', error);
    return null;
  }
}
```
